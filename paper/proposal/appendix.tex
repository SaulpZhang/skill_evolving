\appendix
\section{Detailed Dynamic Pseudo-Regret Analysis}
\label{app:proof}

This appendix proves the dynamic pseudo-regret bound in Section~\ref{sec:theory}. Every bound below is conditional on Assumptions~1--4. In particular, the coverage assumption is essential because the stated result does not hold for an arbitrary sequence of feature vectors.

\subsection{Indexing, filtration, and notation}

The bandit phase is indexed by \(t\in\{1,\ldots,T\}\). To avoid a boundary case in the first \(\ell\) rounds, we regard the \(\ell\) calibration observations from warmup as rounds \(1-\ell,\ldots,0\). By Assumption~4, they are generated with parameter \(\mathbf{W}_1^*\). For every bandit round \(t\), define
\begin{equation}
    \mathcal{W}_t=\{t-\ell,\ldots,t-1\}.
    \label{eq:app-window}
\end{equation}
Thus, \(|\mathcal{W}_t|=\ell\) for every \(t\). This indexing is equivalent to the buffer implementation in Algorithm~\ref{alg:spg}.

Let \(\mathcal{F}_{t-1}\) be the sigma-field generated by the calibration observations and all selected tasks and observed outcomes before bandit round \(t\). For an observed task at index \(i\), abbreviate \(\boldsymbol{\phi}_{i}=\boldsymbol{\phi}_{i,\tau_i}\). Assumption~1 gives
\begin{equation}
    \boldsymbol{\delta}_{i}=\mathbf{W}_i^{*\top}\boldsymbol{\phi}_{i}+\boldsymbol{\varepsilon}_{i},
    \qquad
    \mathbb{E}[\boldsymbol{\varepsilon}_{i}\mid\mathcal{F}_{i-1},\tau_i]=\mathbf{0}.
    \label{eq:app-linear-model}
\end{equation}
For calibration indices \(i\leq0\), set \(\mathbf{W}_i^*=\mathbf{W}_1^*\). Consequently, calibration data do not contribute to the parameter-path variation.

At the beginning of round \(t\), ridge regression uses
\begin{equation}
\begin{aligned}
    \mathbf{A}_{t}&=\lambda\mathbf{I}_{d_f}+\sum_{i\in\mathcal{W}_t}\boldsymbol{\phi}_{i}\boldsymbol{\phi}_{i}^{\top}, \\
    \mathbf{B}_{t}&=\sum_{i\in\mathcal{W}_t}\boldsymbol{\phi}_{i}\boldsymbol{\delta}_{i}^{\top}, \\
    \widehat{\mathbf{W}}_{t}&=\mathbf{A}_{t}^{-1}\mathbf{B}_{t}.
\end{aligned}
\label{eq:app-ridge}
\end{equation}
For a candidate task \(\tau\), write
\begin{equation}
    w_t(\tau)=\lVert\boldsymbol{\phi}_{t,\tau}\rVert_{\mathbf{A}_t^{-1}}
    =\sqrt{\boldsymbol{\phi}_{t,\tau}^{\top}\mathbf{A}_t^{-1}\boldsymbol{\phi}_{t,\tau}}.
    \label{eq:app-width}
\end{equation}

\subsection{A uniform ridge-regression confidence bound}

For a failure probability \(\rho\in(0,1)\), define
\begin{equation}
    \beta_t=\sigma\sqrt{2K\log\!\left(
    \frac{KT\det(\mathbf{A}_t)^{1/2}}{\rho\lambda^{d_f/2}}
    \right)}+\sqrt{\lambda}S.
    \label{eq:app-beta}
\end{equation}
Let
\begin{equation}
    \mathbf{Z}_t=\sum_{i\in\mathcal{W}_t}\boldsymbol{\phi}_{i}\boldsymbol{\varepsilon}_{i}^{\top}.
    \label{eq:app-noise-matrix}
\end{equation}
For the \(k\)-th output coordinate and a fixed window \(\mathcal{W}_t\), the self-normalized martingale inequality gives
\begin{equation}
    \left\lVert\mathbf{A}_t^{-1/2}\mathbf{Z}_t\mathbf{e}_k\right\rVert_2
    \leq\sigma\sqrt{2\log\!\left(
    \frac{KT\det(\mathbf{A}_t)^{1/2}}{\rho\lambda^{d_f/2}}
    \right)}.
    \label{eq:app-coordinate-bound}
\end{equation}
Applying a union bound over the \(K\) output coordinates and the \(T\) windows makes the inequality hold simultaneously for every \(k\in\{1,\ldots,K\}\) and \(t\leq T\), with probability at least \(1-\rho\). The factor \(KT\) inside the logarithm records this union bound. Summing the squared coordinate bounds yields
\begin{equation}
    \left\lVert\mathbf{A}_t^{-1/2}\mathbf{Z}_t\right\rVert_F
    \leq\sigma\sqrt{2K\log\!\left(
    \frac{KT\det(\mathbf{A}_t)^{1/2}}{\rho\lambda^{d_f/2}}
    \right)}.
    \label{eq:app-frobenius-bound}
\end{equation}

\subsection{Prediction error decomposition}

Substituting~\eqref{eq:app-linear-model} into~\eqref{eq:app-ridge} gives
\begin{align}
\mathbf{B}_t
&=\sum_{i\in\mathcal{W}_t}\boldsymbol{\phi}_{i}\boldsymbol{\phi}_{i}^{\top}\mathbf{W}_i^*
+\mathbf{Z}_t.
\label{eq:app-b-expansion}
\end{align}
Since \(\mathbf{A}_t=\lambda\mathbf{I}_{d_f}+\sum_{i\in\mathcal{W}_t}\boldsymbol{\phi}_{i}\boldsymbol{\phi}_{i}^{\top}\), adding and subtracting \(\mathbf{W}_t^*\) gives
\begin{align}
\widehat{\mathbf{W}}_t-\mathbf{W}_t^*
={}&\mathbf{A}_t^{-1}\mathbf{Z}_t-\lambda\mathbf{A}_t^{-1}\mathbf{W}_t^* \nonumber\\
&+\mathbf{A}_t^{-1}\sum_{i\in\mathcal{W}_t}\boldsymbol{\phi}_{i}\boldsymbol{\phi}_{i}^{\top}
\left(\mathbf{W}_i^*-\mathbf{W}_t^*\right).
\label{eq:app-exact-decomposition}
\end{align}
The three terms are the stochastic estimation term, the regularization term, and the stale-window bias.

Fix a candidate task \(\tau\). Cauchy--Schwarz and~\eqref{eq:app-frobenius-bound} give
\begin{align}
\left|\boldsymbol{\phi}_{t,\tau}^{\top}\mathbf{A}_t^{-1}\mathbf{Z}_t\mathbf{g}_t\right|
&\leq w_t(\tau)\left\lVert\mathbf{A}_t^{-1/2}\mathbf{Z}_t\right\rVert_F\left\lVert\mathbf{g}_t\right\rVert_2.
\label{eq:app-noise-projection}
\end{align}
Moreover, since \(\mathbf{A}_t\succeq\lambda\mathbf{I}_{d_f}\),
\begin{align}
\left|\lambda\boldsymbol{\phi}_{t,\tau}^{\top}\mathbf{A}_t^{-1}\mathbf{W}_t^*\mathbf{g}_t\right|
&\leq\sqrt{\lambda}\,S\,w_t(\tau)\left\lVert\mathbf{g}_t\right\rVert_2.
\label{eq:app-regularization-projection}
\end{align}

For the stale-window bias, define
\begin{equation}
    V_t^{\mathrm{win}}=\sum_{i\in\mathcal{W}_t}\left\lVert\mathbf{W}_i^*-\mathbf{W}_t^*\right\rVert_F.
    \label{eq:app-window-variation}
\end{equation}
For every \(i\in\mathcal{W}_t\), Cauchy--Schwarz and \(\mathbf{A}_t\succeq\lambda\mathbf{I}_{d_f}\) imply
\begin{align}
\left|\boldsymbol{\phi}_{t,\tau}^{\top}\mathbf{A}_t^{-1}\boldsymbol{\phi}_{i}\right|
&\leq\left\lVert\mathbf{A}_t^{-1/2}\boldsymbol{\phi}_{t,\tau}\right\rVert_2
\left\lVert\mathbf{A}_t^{-1/2}\boldsymbol{\phi}_{i}\right\rVert_2 \\
&\leq\frac{w_t(\tau)}{\sqrt{\lambda}}.
\label{eq:app-cross-width}
\end{align}
Using \(\lVert\boldsymbol{\phi}_i\rVert_2\leq1\), the bias term in~\eqref{eq:app-exact-decomposition} is bounded by
\begin{equation}
    \frac{w_t(\tau)\lVert\mathbf{g}_t\rVert_2}{\sqrt{\lambda}}V_t^{\mathrm{win}}.
    \label{eq:app-bias-projection}
\end{equation}

Assumption~4 gives \(\mathbf{A}_t\succeq(\lambda+\kappa\ell)\mathbf{I}_{d_f}\), and therefore
\begin{equation}
    w_t(\tau)\leq\frac{1}{\sqrt{\lambda+\kappa\ell}}.
    \label{eq:app-uniform-width}
\end{equation}
Define
\begin{equation}
    D_t=\frac{V_t^{\mathrm{win}}}{\sqrt{\lambda(\lambda+\kappa\ell)}}.
    \label{eq:app-drift-allowance}
\end{equation}
Combining~\eqref{eq:app-exact-decomposition}--\eqref{eq:app-drift-allowance} proves that, simultaneously for all \(t\leq T\) and \(\tau\in\mathcal{T}\),
\begin{equation}
\left|\mathbf{g}_t^{\top}(\widehat{\mathbf{W}}_t-\mathbf{W}_t^*)^{\top}\boldsymbol{\phi}_{t,\tau}\right|
\leq\beta_t\lVert\mathbf{g}_t\rVert_2w_t(\tau)+D_t.
\label{eq:app-uniform-prediction-bound}
\end{equation}

\subsection{Summing the drift allowances}

Let \(v_j=\lVert\mathbf{W}_{j+1}^*-\mathbf{W}_j^*\rVert_F\) for \(j\geq1\), and set \(v_j=0\) for \(j\leq0\). The triangle inequality gives, for \(i<t\),
\begin{equation}
    \left\lVert\mathbf{W}_i^*-\mathbf{W}_t^*\right\rVert_F\leq\sum_{j=i}^{t-1}v_j.
    \label{eq:app-telescoping}
\end{equation}
Thus,
\begin{align}
\sum_{t=1}^{T}V_t^{\mathrm{win}}
&\leq\sum_{t=1}^{T}\sum_{i=t-\ell}^{t-1}\sum_{j=i}^{t-1}v_j \\
&\leq\ell^2\sum_{j=1}^{T-1}v_j
=\ell^2P_T.
\label{eq:app-window-counting}
\end{align}
For each fixed increment \(v_j\), there are at most \(\ell\) windows that contain it and at most \(\ell\) indices within each such window. Combining~\eqref{eq:app-drift-allowance} and~\eqref{eq:app-window-counting} gives
\begin{equation}
    \sum_{t=1}^{T}D_t\leq\frac{\ell^2P_T}{\sqrt{\lambda(\lambda+\kappa\ell)}}.
    \label{eq:app-drift-sum}
\end{equation}

\subsection{From confidence to dynamic pseudo-regret}

Define \(U_t(\tau)=\beta_t\lVert\mathbf{g}_t\rVert_2w_t(\tau)\). Equation~\eqref{eq:app-uniform-prediction-bound} gives, for every candidate task,
\begin{equation}
    q_t(\tau)\leq\mathbf{g}_t^{\top}\widehat{\mathbf{W}}_t^{\top}\boldsymbol{\phi}_{t,\tau}+U_t(\tau)+D_t.
    \label{eq:app-upper-bound}
\end{equation}
The policy selects \(\tau_t\) to maximize the first two terms on the right-hand side. Therefore,
\begin{align}
q_t(\tau_t^*)
&\leq\mathbf{g}_t^{\top}\widehat{\mathbf{W}}_t^{\top}\boldsymbol{\phi}_{t,\tau_t}+U_t(\tau_t)+D_t \\
&\leq q_t(\tau_t)+2U_t(\tau_t)+2D_t.
\label{eq:app-one-step-regret}
\end{align}
The second inequality follows by applying the lower counterpart of~\eqref{eq:app-uniform-prediction-bound} to the selected task.

Since \(\mathbf{g}_t\) is a softmax vector, \(\lVert\mathbf{g}_t\rVert_2\leq1\). Equation~\eqref{eq:app-uniform-width} gives
\begin{equation}
    \sum_{t=1}^{T}U_t(\tau_t)\leq\frac{\beta T}{\sqrt{\lambda+\kappa\ell}},
    \qquad \beta=\max_{t\leq T}\beta_t.
    \label{eq:app-confidence-sum}
\end{equation}
Summing~\eqref{eq:app-one-step-regret}, then applying~\eqref{eq:app-drift-sum} and~\eqref{eq:app-confidence-sum}, establishes
\begin{equation}
    R_T\leq\frac{2\beta T}{\sqrt{\lambda+\kappa\ell}}
    +\frac{2\ell^2P_T}{\sqrt{\lambda(\lambda+\kappa\ell)}}.
    \label{eq:app-final-bound}
\end{equation}

When \(\kappa\ell\geq\lambda\), the two terms in~\eqref{eq:app-final-bound} have orders \(\beta T/(\sqrt{\kappa\ell})\) and \(\ell^{3/2}P_T/(\sqrt{\kappa\lambda})\). For \(P_T>0\), equating these orders gives \(\ell=(\beta T\sqrt{\lambda}/P_T)^{1/2}\). Rounding upward, enforcing \(\kappa\ell\geq\lambda\), and truncating at \(T\) yields the window size in Section~\ref{sec:theory}. If \(P_T=0\), the drift term is zero and the largest feasible window is optimal. \hfill\(\square\)
